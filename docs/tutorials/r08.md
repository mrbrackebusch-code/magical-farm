```ghost
farm.onPlant(function (job) {
    for (let index = 0; index < farm.plantCount(job); index++) {
        farm.plantNext(job)
    }
})

farm.onWater(function (job) {
    for (let index = 0; index < farm.waterCount(job); index++) {
        farm.waterNext(job)
    }
})

farm.onTrellis(function (job) {
    for (let index = 0; index < farm.postCount(job); index++) {
        farm.placePost(job)
        farm.hammerPost(job)
    }
    farm.openGate(job)
})

defenders.onSixShooter(function (job) {
    for (let index = 0; index < defenders.chambers(job); index++) {
        defenders.fireChamber(job)
    }
})

farm.onBaskets(function (job) {
    for (let index = 0; index < farm.basketCount(job); index++) {
        farm.pickClump(job)
        farm.carryBasket(job)
        farm.depositBasket(job)
    }
})

defenders.onTwinbud(function (job) {
    for (let index = 0; index < defenders.twinCycles(job); index++) {
        defenders.fireLeft(job)
        defenders.fireRight(job)
    }
})

farm.onBlooms(function (job) {
    let power = 1
    for (let index = 0; index < farm.bloomCount(job); index++) {
        farm.growBloom(job, power)
        power += 1
    }
})

defenders.onMineberry(function (job) {
    for (let index = 0; index < defenders.trapCount(job); index++) {
        defenders.nextTrap(job)
        defenders.dropMine(job)
    }
})

farm.onWeeds(function (job) {
    for (let index = 0; index < farm.weedPlotCount(job); index++) {
        if (farm.weedsHere(job)) {
            farm.clearWeeds(job)
        }
        farm.plantCurrent(job)
        farm.waterCurrent(job)
    }
})

defenders.onRage(function (job) {
    let damage = defenders.startingDamage(job)
    while (defenders.enemyNearby(job)) {
        defenders.strike(job, damage)
        damage += 1
    }
})

farm.onIrrigate(function (job) {
    let water = farm.waterSupply(job)
    while (water > 0) {
        farm.waterNext(job)
        water -= 1
    }
})
```

## Dry Terrace: let the remaining water decide when to stop

The Dry Terrace does not always start with the same amount of water. The same code must handle 3 units, then an empty supply, then 4 units after refill.

## Change the state that controls the while loop

In ``||farm(noclick):Farm jobs||``, find ``||farm(noclick):on irrigation requested||``, ``||farm(noclick):water available at start for job job||``, and ``||farm(noclick):water next dry plot for job job||``. Make a variable named `water` in ``||variables(noclick):Variables||`` and find ``||loops(noclick):while||`` in ``||loops(noclick):Loops||``.

Set `water` once to **water available at start for job job**. That reporter is the starting snapshot; `water` is the learner-owned value your loop will change.

Use `while water > 0`. Inside the loop, water the next dry plot, then change `water` by `-1`. Do not reread the starting-supply reporter as though it were the changing loop variable.

Run the full teaching sequence without editing your code between runs: 3 units should water three plots; 0 units should run the body zero times; after the spring refills to 4, the same code should water four more plots.

If the loop keeps trying after the physical water is gone, inspect whether your learner variable actually decreases. The world will show dry attempts or safety handling rather than secretly repairing the code.
