```ghost
farm.onPlant(function (job) {
    for (let index = 0; index < farm.plantCount(job); index++) {

    }
})

farm.onWater(function (job) {
    for (let index = 0; index < farm.waterCount(job); index++) {
        farm.waterNext(job)
    }
})

farm.onTrellis(function (job) {
    for (let index = 0; index < farm.postCount(job); index++) {
        farm.placePost(job)
        farm.hammerPost(job)
    }
    farm.openGate(job)
})

defenders.onSixShooter(function (job) {
    for (let index = 0; index < defenders.chambers(job); index++) {
        defenders.fireChamber(job)
    }
})

farm.onBaskets(function (job) {
    for (let index = 0; index < farm.basketCount(job); index++) {
        farm.pickClump(job)
        farm.carryBasket(job)
        farm.depositBasket(job)
    }
})

defenders.onTwinbud(function (job) {
    for (let index = 0; index < defenders.twinCycles(job); index++) {
        defenders.fireLeft(job)
        defenders.fireRight(job)
    }
})

farm.onBlooms(function (job) {
    let power = 1
    for (let index = 0; index < farm.bloomCount(job); index++) {
        farm.growBloom(job, power)
        power += 1
    }
})

defenders.onMineberry(function (job) {
    for (let index = 0; index < defenders.trapCount(job); index++) {
        defenders.nextTrap(job)
        defenders.dropMine(job)
    }
})

farm.onWeeds(function (job) {
    for (let index = 0; index < farm.weedPlotCount(job); index++) {
        if (farm.weedsHere(job)) {
            farm.clearWeeds(job)
        }
        farm.plantCurrent(job)
        farm.waterCurrent(job)
    }
})

defenders.onRage(function (job) {
    let damage = defenders.startingDamage(job)
    while (defenders.enemyNearby(job)) {
        defenders.strike(job, damage)
        damage += 1
    }
})

farm.onIrrigate(function (job) {
    let water = farm.waterSupply(job)
    while (water > 0) {
        farm.waterNext(job)
        water -= 1
    }
})

defenders.onPrism(function (job) {
    for (let index = 0; index < defenders.markedCount(job); index++) {
        defenders.inspectEnemy(job)
        if (defenders.armored(job)) {
            defenders.pierce(job)
        } else {
            defenders.splash(job)
        }
    }
})

farm.onFertilize(function (job) {
    for (let index = 0; index < farm.treeCount(job); index++) {
        farm.fertilizeTree(job, index + 1)
    }
})

defenders.onSunbeam(function (job) {
    let sunlight = defenders.sunlightSupply(job)
    while (sunlight > 0) {
        defenders.beam(job)
        sunlight -= 1
```
```blocks
    }
})
```

## Use the native index

In ``||farm(noclick):Farm jobs||``, find ``||farm(noclick):on orchard work requested||``, ``||farm(noclick):number of orchard trees for job job||``, and ``||farm(noclick):fertilize next tree with value for job job||``. Find the native indexed loop in ``||loops(noclick):Loops||``.

The indexed loop has a counter variable built in. In MakeCode Blocks its first value is `0`, and that value changes automatically each iteration.

Loop from the native zero-based `index` while it is below **number of orchard trees for job job**. Give the fertilizer action `index + 1` so the visible tree levels are 1, 2, 3, 4 rather than 0, 1, 2, 3.

Do not create a second counter. Run Orchard Lane and watch the values on the trees: baseline should be exactly 1 → 2 → 3 → 4.

## use the resource while pattern for Sunbeam Lily

At Defense Prep, Sunbeam Lily uses a pattern you already built at the Dry Terrace. In ``||defenders(noclick):Defender powers||``, find ``||defenders(noclick):on Sunbeam Lily power requested||``, ``||defenders(noclick):sunlight available at start for job job||``, and ``||defenders(noclick):beam for job job||``.

Make a learner variable named `sunlight`. Copy the starting sunlight reporter into it once. Use `while sunlight > 0`. Inside the loop, beam once, then change `sunlight` by `-1`.

A full baseline resource can produce 6 beams. The optional upgrade can raise capacity to 8, but an invocation may also begin partly depleted or at zero. The same code should respect the actual starting value reported by the world.
