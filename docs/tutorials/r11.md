```ghost
farm.onPlant(function (job) {

    }
})

farm.onWater(function (job) {
    for (let index = 0; index < farm.waterCount(job); index++) {
        farm.waterNext(job)
    }
})

farm.onTrellis(function (job) {
    for (let index = 0; index < farm.postCount(job); index++) {
        farm.placePost(job)
        farm.hammerPost(job)
    }
    farm.openGate(job)
})

defenders.onSixShooter(function (job) {
    for (let index = 0; index < defenders.chambers(job); index++) {
        defenders.fireChamber(job)
    }
})

farm.onBaskets(function (job) {
    for (let index = 0; index < farm.basketCount(job); index++) {
        farm.pickClump(job)
        farm.carryBasket(job)
        farm.depositBasket(job)
    }
})

defenders.onTwinbud(function (job) {
    for (let index = 0; index < defenders.twinCycles(job); index++) {
        defenders.fireLeft(job)
        defenders.fireRight(job)
    }
})

farm.onBlooms(function (job) {
    let power = 1
    for (let index = 0; index < farm.bloomCount(job); index++) {
        farm.growBloom(job, power)
        power += 1
    }
})

defenders.onMineberry(function (job) {
    for (let index = 0; index < defenders.trapCount(job); index++) {
        defenders.nextTrap(job)
        defenders.dropMine(job)
    }
})

farm.onWeeds(function (job) {
    for (let index = 0; index < farm.weedPlotCount(job); index++) {
        if (farm.weedsHere(job)) {
            farm.clearWeeds(job)
        }
        farm.plantCurrent(job)
        farm.waterCurrent(job)
    }
})

defenders.onRage(function (job) {
    let damage = defenders.startingDamage(job)
    while (defenders.enemyNearby(job)) {
        defenders.strike(job, damage)
        damage += 1
    }
})

farm.onIrrigate(function (job) {
    let water = farm.waterSupply(job)
    while (water > 0) {
        farm.waterNext(job)
        water -= 1
    }
})

defenders.onPrism(function (job) {
    for (let index = 0; index < defenders.markedCount(job); index++) {
        defenders.inspectEnemy(job)
        if (defenders.armored(job)) {
            defenders.pierce(job)
        } else {
            defenders.splash(job)
        }
    }
})

farm.onFertilize(function (job) {
    for (let index = 0; index < farm.treeCount(job); index++) {
        farm.fertilizeTree(job, index + 1)
    }
})

defenders.onSunbeam(function (job) {
    let sunlight = defenders.sunlightSupply(job)
    while (sunlight > 0) {
        defenders.beam(job)
        sunlight -= 1
    }
})

farm.onFeed(function (job) {
    for (let index = 0; index < farm.animalCount(job); index++) {
        farm.inspectAnimal(job)
        if (farm.hungry(job)) {
            farm.feedAnimal(job)
        }
    }
})

defenders.onEcho(function (job) {
    for (let index = 0; index < defenders.pulseCount(job); index++) {
        defenders.pulse(job, index + 1)
    }
})
```

Six baseline animals show mixed states. Every animal should be checked, but a full animal should not be fed again.

## use repeat + inspect + if

In ``||farm(noclick):Farm jobs||``, find ``||farm(noclick):on animal feeding requested||``, ``||farm(noclick):number of animals for job job||``, ``||farm(noclick):inspect next animal for job job||``, ``||farm(noclick):current animal hungry for job job||``, and ``||farm(noclick):feed current animal for job job||``.

Reuse a structure you already know: repeat through the supplied count, inspect first, and put the feed action inside an `if` that tests the current inspected animal.

There is no `else` action here. If an animal is already full, doing nothing is the correct false case.

Run Animal Meadow. All six baseline animals should be inspected, and only the HUNGRY animals should visibly become fed.

## use the index for Echo Mushroom

At Defense Prep, Echo Mushroom reuses Orchard Lane's index idea. In ``||defenders(noclick):Defender powers||``, find ``||defenders(noclick):on Echo Mushroom power requested||``, ``||defenders(noclick):number of pulse levels for job job||``, and ``||defenders(noclick):pulse radius value for job job||``.

Use the native zero-based `index` again. The visible pulse radius should be `index + 1`, so the baseline sequence is 1 → 2 → 3 → 4 → 5.

Use the named pulse-level reporter as the bound. The optional upgrade adds radius 6 without requiring a code edit.
