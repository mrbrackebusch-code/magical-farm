```ghost
        farm.depositBasket(job)
    }
})

defenders.onTwinbud(function (job) {
    for (let index = 0; index < defenders.twinCycles(job); index++) {
        defenders.fireLeft(job)
        defenders.fireRight(job)
    }
})

farm.onBlooms(function (job) {
    let power = 1
    for (let index = 0; index < farm.bloomCount(job); index++) {
        farm.growBloom(job, power)
        power += 1
    }
})

defenders.onMineberry(function (job) {
    for (let index = 0; index < defenders.trapCount(job); index++) {
        defenders.nextTrap(job)
        defenders.dropMine(job)
    }
})

farm.onWeeds(function (job) {
    for (let index = 0; index < farm.weedPlotCount(job); index++) {
        if (farm.weedsHere(job)) {
            farm.clearWeeds(job)
        }
        farm.plantCurrent(job)
        farm.waterCurrent(job)
    }
})

defenders.onRage(function (job) {
    let damage = defenders.startingDamage(job)
    while (defenders.enemyNearby(job)) {
        defenders.strike(job, damage)
        damage += 1
    }
})

farm.onIrrigate(function (job) {
    let water = farm.waterSupply(job)
    while (water > 0) {
        farm.waterNext(job)
        water -= 1
    }
})

defenders.onPrism(function (job) {
    for (let index = 0; index < defenders.markedCount(job); index++) {
        defenders.inspectEnemy(job)
        if (defenders.armored(job)) {
            defenders.pierce(job)
        } else {
            defenders.splash(job)
        }
    }
})
```

## Prism Bloom: choose one of two attacks

Each marked enemy needs exactly one attack choice. Armored enemies need piercing pollen; unarmored enemies need splash pollen.

## Add the false case with else

You already used `if` to decide whether a plot needed clearing. An ``||logic(noclick):else||`` branch handles the false case of that same test: if the test is true, the `if` branch runs; otherwise the `else` branch runs.

In ``||defenders(noclick):Defender powers||``, find ``||defenders(noclick):on Prism Bloom power requested||``, ``||defenders(noclick):number of marked targets for job job||``, ``||defenders(noclick):inspect next marked enemy for job job||``, ``||defenders(noclick):current enemy armored for job job||``, ``||defenders(noclick):piercing pollen for job job||``, and ``||defenders(noclick):splash pollen for job job||``.

Repeat once for each **number of marked targets for job job**. Inside the repeat, inspect the next marked enemy **before** testing whether the current enemy is armored.

If the current enemy is armored, use piercing pollen. Use the `else` branch for splash pollen. Run the preview and check both target types.

At baseline the preview has 4 marked targets. The optional marked-target upgrade changes that reporter to 5, so the same loop should inspect one more target without a code edit.
