```ghost
    }

})

defenders.onSixShooter(function (job) {

        defenders.fireChamber(job)
    }
})

farm.onBaskets(function (job) {
    for (let index = 0; index < farm.basketCount(job); index++) {
        farm.pickClump(job)
        farm.carryBasket(job)
        farm.depositBasket(job)
    }
})

defenders.onTwinbud(function (job) {
    for (let index = 0; index < defenders.twinCycles(job); index++) {
        defenders.fireLeft(job)
        defenders.fireRight(job)
    }
})

farm.onBlooms(function (job) {
    let power = 1
    for (let index = 0; index < farm.bloomCount(job); index++) {
        farm.growBloom(job, power)
        power += 1
    }
})

defenders.onMineberry(function (job) {
    for (let index = 0; index < defenders.trapCount(job); index++) {
        defenders.nextTrap(job)
        defenders.dropMine(job)
    }
})

farm.onWeeds(function (job) {
    for (let index = 0; index < farm.weedPlotCount(job); index++) {
        if (farm.weedsHere(job)) {
            farm.clearWeeds(job)
        }
        farm.plantCurrent(job)
        farm.waterCurrent(job)
    }
})

defenders.onRage(function (job) {
    let damage = defenders.startingDamage(job)
    while (defenders.enemyNearby(job)) {
        defenders.strike(job, damage)
        damage += 1
    }
})

farm.onIrrigate(function (job) {
    let water = farm.waterSupply(job)
    while (water > 0) {
        farm.waterNext(job)
        water -= 1
    }
})

defenders.onPrism(function (job) {
    for (let index = 0; index < defenders.markedCount(job); index++) {
        defenders.inspectEnemy(job)
        if (defenders.armored(job)) {
            defenders.pierce(job)
        } else {
            defenders.splash(job)
        }
    }
})

farm.onFertilize(function (job) {
    for (let index = 0; index < farm.treeCount(job); index++) {
        farm.fertilizeTree(job, index + 1)
    }
})

defenders.onSunbeam(function (job) {
    let sunlight = defenders.sunlightSupply(job)
    while (sunlight > 0) {
        defenders.beam(job)
        sunlight -= 1
    }
})

farm.onFeed(function (job) {
    for (let index = 0; index < farm.animalCount(job); index++) {
        farm.inspectAnimal(job)
        if (farm.hungry(job)) {
            farm.feedAnimal(job)
        }
    }
})

defenders.onEcho(function (job) {
    for (let index = 0; index < defenders.pulseCount(job); index++) {
        defenders.pulse(job, index + 1)
    }
})

farm.onGreenhouse(function (job) {
    for (let bed = 0; bed < farm.greenhouseBeds(job); bed++) {
        farm.nextBed(job)
        for (let plot = 0; plot < farm.plotsPerBed(job); plot++) {
            farm.plantNext(job)
        }
    }
})

defenders.onHealroot(function (job) {
    for (let index = 0; index < defenders.allyCount(job); index++) {
        defenders.inspectAlly(job)
        if (defenders.damagedAlly(job)) {
            defenders.heal(job)
        }
    }
})
```

## Greenhouse Grid: repeat a whole repeated job

The greenhouse is visibly grouped into 3 baseline beds with 4 plots in each bed. The important structure is not just twelve plant actions: finish one bed's four plots, then repeat that bed-sized job for the next bed.

## Build the grouped work

With 3 beds and 4 plots in each bed, **plant next plot for job job** runs 12 times altogether. Each group of four plant actions belongs to one bed.

In ``||farm(noclick):Farm jobs||``, find ``||farm(noclick):on greenhouse work requested||``, ``||farm(noclick):number of greenhouse beds for job job||``, ``||farm(noclick):plots in each greenhouse bed for job job||``, ``||farm(noclick):move to next greenhouse bed for job job||``, and ``||farm(noclick):plant next plot for job job||``.

Start with one bed-sized job: move to the next bed once, then repeat planting for every plot in that bed. Now put that whole job inside an outer repeat for the number of greenhouse beds.

Run the Greenhouse Grid. Baseline should show four completed plots in one bed before the farm advances to the next bed. A flat total of twelve plant calls is not the same structure, because the supplied world keeps bed context.

## Use inspect + if to Healroot

At Defense Prep, Healroot uses the same inspect-and-decide pattern from Animal Meadow. In ``||defenders(noclick):Defender powers||``, find ``||defenders(noclick):on Healroot power requested||``, ``||defenders(noclick):allies to inspect for job job||``, ``||defenders(noclick):inspect next ally for job job||``, ``||defenders(noclick):current ally damaged for job job||``, and ``||defenders(noclick):heal current ally for job job||``.

Repeat through the supplied ally count, inspect first, and heal only when the current ally is damaged. Healthy allies are intentionally skipped.

Baseline inspects 4 allies. The optional upgrade changes the named reporter to 5, so the unchanged algorithm can inspect one more candidate.
