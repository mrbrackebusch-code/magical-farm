```ghost
        farm.waterNext(job)
    }
})

farm.onTrellis(function (job) {
    for (let index = 0; index < farm.postCount(job); index++) {
        farm.placePost(job)
        farm.hammerPost(job)
    }
    farm.openGate(job)
})

defenders.onSixShooter(function (job) {
    for (let index = 0; index < defenders.chambers(job); index++) {
        defenders.fireChamber(job)
    }
})

farm.onBaskets(function (job) {
    for (let index = 0; index < farm.basketCount(job); index++) {
        farm.pickClump(job)
        farm.carryBasket(job)
        farm.depositBasket(job)
    }
})

defenders.onTwinbud(function (job) {
    for (let index = 0; index < defenders.twinCycles(job); index++) {
        defenders.fireLeft(job)
        defenders.fireRight(job)
    }
})

farm.onBlooms(function (job) {
    let power = 1
    for (let index = 0; index < farm.bloomCount(job); index++) {
        farm.growBloom(job, power)
        power += 1
    }
})

defenders.onMineberry(function (job) {
    for (let index = 0; index < defenders.trapCount(job); index++) {
        defenders.nextTrap(job)
        defenders.dropMine(job)
    }
})

farm.onWeeds(function (job) {
    for (let index = 0; index < farm.weedPlotCount(job); index++) {
        if (farm.weedsHere(job)) {
            farm.clearWeeds(job)
        }
        farm.plantCurrent(job)
        farm.waterCurrent(job)
    }
})
```

## Overgrown Field: not every plot needs the same first step

Eight baseline plots need planting and watering. Some are overgrown and need clearing first; clean plots should skip clearing and still finish the rest of the work.

## Put a decision inside the repeat

In ``||farm(noclick):Farm jobs||``, find ``||farm(noclick):on overgrown field work requested||``, ``||farm(noclick):field plots for job job||``, ``||farm(noclick):weeds here for job job||``, ``||farm(noclick):clear weeds here for job job||``, ``||farm(noclick):plant current plot for job job||``, and ``||farm(noclick):water current plot for job job||``. Find ``||logic(noclick):if||`` in ``||logic(noclick):Logic||`` and ``||loops(noclick):repeat||`` in ``||loops(noclick):Loops||``.

Use **field plots for job job** as the repeat count. Inside that repeat, test **weeds here for job job**. Only **clear weeds here for job job** belongs inside the `if`.

After the `if`, but still inside the repeat, plant the current plot and water the current plot. That way a clean plot skips clearing without skipping the work every plot still needs.

Run the Overgrown Field. All eight baseline plots should finish. If clean plots stay unfinished, look at which blocks accidentally became part of the `if`.
