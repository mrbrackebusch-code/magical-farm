```ghost
farm.onPlant(function (job) {
    for (let index = 0; index < farm.plantCount(job); index++) {
        farm.plantNext(job)
    }
})

farm.onWater(function (job) {
    for (let index = 0; index < farm.waterCount(job); index++) {
        farm.waterNext(job)
    }
})

farm.onTrellis(function (job) {
    for (let index = 0; index < farm.postCount(job); index++) {
        farm.placePost(job)
        farm.hammerPost(job)
    }
    farm.openGate(job)
})

defenders.onSixShooter(function (job) {
    for (let index = 0; index < defenders.chambers(job); index++) {
        defenders.fireChamber(job)
    }
})

farm.onBaskets(function (job) {
    for (let index = 0; index < farm.basketCount(job); index++) {
        farm.pickClump(job)
        farm.carryBasket(job)
        farm.depositBasket(job)
    }
})

defenders.onTwinbud(function (job) {
    for (let index = 0; index < defenders.twinCycles(job); index++) {
        defenders.fireLeft(job)
        defenders.fireRight(job)
    }
})

farm.onBlooms(function (job) {
    let power = 1
    for (let index = 0; index < farm.bloomCount(job); index++) {
        farm.growBloom(job, power)
        power += 1
    }
})

defenders.onMineberry(function (job) {
    for (let index = 0; index < defenders.trapCount(job); index++) {
        defenders.nextTrap(job)
        defenders.dropMine(job)
    }
})

farm.onWeeds(function (job) {
    for (let index = 0; index < farm.weedPlotCount(job); index++) {
        if (farm.weedsHere(job)) {
            farm.clearWeeds(job)
        }
        farm.plantCurrent(job)
        farm.waterCurrent(job)
    }
})

defenders.onRage(function (job) {
    let damage = defenders.startingDamage(job)
    while (defenders.enemyNearby(job)) {
        defenders.strike(job, damage)
        damage += 1
    }
})
```

## Rage Orchid: repeat for as long as the situation lasts

Rage Orchid should keep striking while a valid enemy remains nearby. Unlike the counted repeats you already know, you do not know ahead of time how many strikes the engagement will need.

## Build your first while loop

A ``||loops(noclick):while||`` loop keeps checking a condition and repeats while that condition remains true. Here, the condition is supplied directly by the battle: **enemy nearby for job job**.

In ``||defenders(noclick):Defender powers||``, find ``||defenders(noclick):on Rage Orchid power requested||``, ``||defenders(noclick):starting strike damage for job job||``, ``||defenders(noclick):enemy nearby for job job||``, and ``||defenders(noclick):strike with damage value for job job||``. In ``||variables(noclick):Variables||``, make a variable named `damage`.

Set `damage` to **starting strike damage for job job** before the while loop. Use **enemy nearby for job job** as the while condition. Inside the loop, strike with the current `damage`, then change `damage` by `1`.

Run the Rage preview. At baseline the visible damage should begin at 1 and rise with each strike until the enemy is no longer nearby. The optional starting-damage upgrade begins at 2 without changing this algorithm.
